def request_post(url, headers, data):
    
    res = requests.post(
        f'https://openapi.koreainvestment.com:9443{url}',
        headers = headers,
        json = data,
        timeout = 10
    )
    res.raise_for_status()

    j = res.json()
    if j.get("rt_cd") not in (None, "0"):
        raise RuntimeError(f"KIS error: {j.get('msg_cd')} {j.get('msg1')}")

    return j


# 접근 토큰 발급
### url : /oauth2/tokenP
def _issue_oauth_token(app_key, app_secret):
    
    url = '/oauth2/tokenP'
    
    headers = {
        'Content-Type': 'application/json'
    }
    
    data = {
        'grant_type': 'client_credentials',
        'appkey': app_key,
        'appsecret': app_secret
    }

    res = request_post(url, headers, data)

    return res
